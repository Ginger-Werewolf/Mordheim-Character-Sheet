<script type="text/worker">

  // Your Experience tracker code goes here
  
(function () {
  function xpAttribute(number) {
    return "xp_" + String(number).padStart(2, "0");
  }

  function handleXPChange(changedNumber, eventInfo) {
    // Ignore changes made by this worker.
    if (eventInfo.sourceType === "sheetworker") {
      return;
    }

    var attributes = [];

    for (var i = 1; i <= 90; i++) {
      attributes.push(xpAttribute(i));
    }

    getAttrs(attributes, function (values) {
      var updates = {};
      var changedValue = values[xpAttribute(changedNumber)];

      if (changedValue === "1") {
        // Check that every earlier box is ticked.
        for (var i = 1; i < changedNumber; i++) {
          if (values[xpAttribute(i)] !== "1") {
            updates[xpAttribute(changedNumber)] = "0";
            break;
          }
        }
      } else {
        // Unticking a box clears every later box.
        for (var i = changedNumber + 1; i <= 90; i++) {
          if (values[xpAttribute(i)] === "1") {
            updates[xpAttribute(i)] = "0";
          }
        }
      }

      if (Object.keys(updates).length > 0) {
        setAttrs(updates);
      }
    });
  }

  // Watch all 90 Experience boxes.
  for (var i = 1; i <= 90; i++) {
    (function (number) {
      on("change:" + xpAttribute(number), function (eventInfo) {
        handleXPChange(number, eventInfo);
      });
    })(i);
  }
})();


</script>
